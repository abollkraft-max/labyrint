// Anslut till Python-servern via datorns lokala IP-adress const ws = new WebSocket('ws://192.168.0.16:8765'); ws.onopen = () => { console.log("Ansluten till Python-servern!"); }; ws.onclose = () => { console.log("Anslutningen bröts."); }; ws.onerror = (error) => { console.error("WebSocket-fel:", error); }; // Funktion för att hantera rörelsedata från telefonens gyro function handleOrientation(event) { const data = { alpha: event.alpha, // Rotation kring z-axeln beta: event.beta, // Lutning framåt/bakåt (x-axeln) gamma: event.gamma // Lutning åt sidorna (y-axeln) }; // Skicka datan till Python-servern om anslutningen är öppen if (ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify(data)); } } // Startknapp för att begära tillgång till sensorer (krävs på nya mobiler) const startButton = document.getElementById('startButton'); if (startButton) { startButton.addEventListener('click', () => { if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') { DeviceOrientationEvent.requestPermission() .then(permissionState => { if (permissionState === 'granted') { window.addEventListener('deviceorientation', handleOrientation); console.log("Rörelsesensorer aktiverade!"); } else { alert("Behöver tillgång till rörelsesensorer för att funka."); } }) .catch(console.error); } else { window.addEventListener('deviceorientation', handleOrientation); console.log("Rörelsesensorer lyssnar direkt."); } }); }
